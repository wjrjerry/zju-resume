\section{Research Projects}

\projectheading{AuraForge: Beyond Human-Written Tests for Training Secure Vibe Coding Agents}{Jun. 2026 -- Present}
\affiliation{Li Lab}
\begin{itemize}
    \item \textbf{Research objective:} Develop scalable executable supervision for training coding agents to satisfy \textbf{implicit security requirements}, evaluating generated patches against both functional behavior and vulnerability-specific security properties.
    \item \textbf{Task construction:} Led the \textbf{JavaScript/TypeScript task-construction pipeline} for converting real-world vulnerability fixes into executable coding-agent tasks. Designed \textbf{language-aware feature masking} to remove vulnerable implementations while preserving declarations, exports, type signatures, and interfaces for independent reimplementation.
    \item \textbf{Execution infrastructure and scale:} Reconstructed \textbf{repository-specific Node.js environments}, including runtime, package-manager, workspace, dependency, and build configurations. Developed \textbf{test adapters that unify heterogeneous JS/TS test runners} for consistent functional and security validation across repository states. Processed \textbf{863 curated JS/TS candidates into 229 task instances} for secure-agent evaluation and training.
\end{itemize}

\projectsep

\projectheading{SeekJudge: Reward Modeling for Computer-Use Agents}{Mar. 2026 -- Present}
\affiliation{ZJU Scaling Logger (ZJUSCL)}
\begin{itemize}
    \item \textbf{Research objective:} Developed model-based rewards for long-horizon computer-use agents, enabling online RL without task-specific rule-based verifiers.
    \item \textbf{Method:} Co-developed the \textbf{Condense--Ground--Seek--Analyze} framework, compressing long screenshot--action trajectories into semantic timelines, grounding low-level actions, and iteratively localizing and inspecting decisive visual evidence through a Seek--Analyze loop. The resulting multi-agent judging process was distilled into a unified \textbf{9B judge model}.
\end{itemize}

\projectsep

\projectheading{TradeActor: Multimodal Market Reasoning for Portfolio Allocation}{Mar. 2026 -- May 2026}
\begin{itemize}
    \item Developed a \textbf{multimodal trading framework} that combines stock-price and technical-indicator charts with market context, using a \textbf{frozen vision-language model} to extract semantic signals for downstream portfolio decisions.
    \item Designed a lightweight \textbf{Qwen-0.6B portfolio actor} that encodes market context into a designated \textbf{forecast token} and projects its final hidden state through an \textbf{MLP action head} into continuous portfolio allocations.
    \item Optimized the policy for \textbf{portfolio returns under transaction-cost penalties} and evaluated daily allocation decisions through \textbf{end-to-end backtesting on StockBench}.
\end{itemize}

\projectsep

\projectheading{TimeThinkingLLM: Multimodal Time-Series Reasoning}{Sep. 2025 -- Jan. 2026}
\begin{itemize}
    \item Developed a \textbf{multimodal framework} that aligns \textbf{continuous time-series signals} with the \textbf{representation space of Qwen-3}, enabling a pretrained language model to directly interpret temporal data.
    \item Designed a \textbf{temporal encoder} and \textbf{projection mechanism} to transform time-series signals into representations compatible with the LLM's \textbf{semantic embedding space}.
    \item Applied \textbf{GRPO-based reinforcement learning} to train the model to \textbf{reason over temporal signals}, and evaluated its temporal reasoning capabilities on \textbf{time-series benchmarks}.
\end{itemize}
