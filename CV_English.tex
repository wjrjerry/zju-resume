\section{Research Projects}

\projectheading{AuraForge: Secure Coding-Agent Evaluation and Training}{Jun. 2026 -- Present}
\affiliation{Li Lab}
\begin{itemize}
    \item \textbf{Research objective:} Evaluate repository-level patches with functional and security tests; investigate \textbf{safety monitors} that identify execution risks and guide agent decisions to improve security while preserving functional correctness.
    \item \textbf{Contribution:} Built a task-construction pipeline that extracts vulnerability-fixing commits from public GitHub Security Advisories (GHSAs) and converts them into repository-level tasks with functional and security tests for \textbf{joint evaluation of correctness and patch security}.
\end{itemize}

\projectsep

\projectheading{SeekJudge: Reward Modeling for Computer-Use Agents}{Mar. 2026 -- May 2026}
\affiliation{ZJU Scaling Logger (ZJUSCL)}
\begin{itemize}
    \item \textbf{Research problem:} Assess long-horizon GUI task completion to provide model-based rewards for agent RL. The project distills multi-agent judging into a \textbf{9B judge model} that matches or surpasses rule-based rewards in online RL.
    \item \textbf{Contribution:} Developed the Condense--Ground--Seek--Analyze judging pipeline, using an \textbf{iterative Seek--Analyze loop} to gather evidence from long screenshot and action sequences and determine task completion.
    \item \textbf{System implementation:} Built the trajectory-processing and \textbf{HTTP reward-serving pipeline}, supporting offline batch evaluation and online RL.
\end{itemize}

\projectsep

\projectheading{TradeActor: Multimodal Stock Forecasting and Portfolio Learning}{Mar. 2026 -- May 2026}
\begin{itemize}
    \item Combined stock charts, numerical data, and news to generate and evaluate \textbf{LLM-based trading signals} for DOW-20; trained a \textbf{Qwen-based portfolio policy} with return and transaction-cost objectives.
    \item Integrated the policy into \textbf{StockBench} for daily allocation decisions and end-to-end portfolio backtesting.
\end{itemize}

\projectheading{TimeThinkingLLM: Multimodal Time-Series Reasoning}{Sep. 2025 -- Jan. 2026}
\begin{itemize}
